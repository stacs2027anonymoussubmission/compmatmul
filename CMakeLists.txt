# CMake file for building compmatmul

cmake_minimum_required(VERSION 4.3)

project(compmatmul)

set(CMAKE_CXX_STANDARD 26)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    message(WARNING "No build type specified, defaulting to Debug")
    set(CMAKE_BUILD_TYPE
        Debug
        CACHE STRING "Build type"
        FORCE)
    set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS
        Debug Release)
endif()

if(NOT DEFINED ARCH)
  message(WARNING "No ARCH, CPU, or TUNE defined, assuming -march=native (specify -DARCH, -DCPU, and -DTUNE explicitly for maximum performance if running on another system)")
  set(ARCH native CACHE STRING "Which -march option to use?")
  set(CPU native CACHE STRING "Which -mcpu option to use? (ignored on x86_64)")
  set(TUNE native CACHE STRING "Which -mtune option to use?")
endif()

option(BUILD_TESTS "Build the unit tests" ON)
option(BUILD_DOCS "Build the documentation" OFF)
option(BUILD_MODULE "Build the Python module" ON)
option(BUILD_BENCHMARK "Build the benchmark tool" ON)
option(BUILD_MULTIPLY "Build the multiply tool" ON)
option(BUILD_SYNTHETIC "Build the synthetic benchmark tool" ON)

add_library(libarray INTERFACE libarray/random.hpp libarray/array.hpp)

set(BLAS_BACKEND "Generic" CACHE STRING "Which BLAS backend to use")
set_property(CACHE BLAS_BACKEND PROPERTY STRINGS Generic OpenBLAS MKL Accelerate AOCL ARMPL)

set(BLAS_BACKENDS Generic OpenBLAS MKL Accelerate AOCL ARMPL)
if(NOT BLAS_BACKEND IN_LIST BLAS_BACKENDS)
    message(FATAL_ERROR "Invalid BLAS backend specified")
endif()

if(BLAS_BACKEND STREQUAL "Generic")
    unset(BLA_VENDOR)
    target_compile_definitions(libarray INTERFACE LIBARRAY_BLAS_GENERIC)
elseif(BLAS_BACKEND STREQUAL "OpenBLAS")
    message(FATAL_ERROR "OpenBLAS backend is not supported yet, please use MKL or Accelerate")
    target_compile_definitions(libarray INTERFACE LIBARRAY_BLAS_OPENBLAS)
elseif(BLAS_BACKEND STREQUAL "MKL")
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
        target_compile_definitions(libarray INTERFACE LIBARRAY_BLAS_MKL MKL_ILP64)
        set(MKL_INTERFACE "ilp64" CACHE STRING "Which MKL interface to use" FORCE)
        set(MKL_LINK "dynamic" CACHE STRING "Which MKL LINKAGE to use" FORCE)
        set(MKL_THREADING "gnu_thread" CACHE STRING "Which MKL threading to use" FORCE)
        find_package(MKL CONFIG REQUIRED)
        target_link_libraries(libarray INTERFACE MKL::MKL)
    else()
        message(FATAL_ERROR "MKL backend is only supported on x86_64 architectures")
    endif()
elseif(BLAS_BACKEND STREQUAL "AOCL")
    message(FATAL_ERROR "AOCL backend is not supported yet, please use MKL or Accelerate")
    target_compile_definitions(libarray INTERFACE LIBARRAY_BLAS_AOCL)
elseif(BLAS_BACKEND STREQUAL "Accelerate")
    if(APPLE)
        set(BLA_VENDOR Apple)
        target_compile_definitions(libarray INTERFACE LIBARRAY_BLAS_ACCELERATE 
            ACCELERATE_LAPACK_ILP64 ACCELERATE_NEW_LAPACK)
    else()
        message(FATAL_ERROR "Accelerate backend is only supported on macOS")
    endif()
elseif(BLAS_BACKEND STREQUAL "ARMPL")
    message(FATAL_ERROR "ARMPL backend is not supported yet, please use MKL or Accelerate")
    target_compile_definitions(libarray INTERFACE LIBARRAY_BLAS_ARMPL)
else()
    message(FATAL_ERROR "Invalid BLAS backend specified")
endif()

if(NOT BLAS_BACKEND STREQUAL "MKL")
    find_package(BLAS REQUIRED)
    target_link_libraries(libarray INTERFACE BLAS::BLAS)
endif()

set(FFT_BACKEND "None" CACHE STRING "Which FFT backend to use")
set_property(CACHE FFT_BACKEND PROPERTY STRINGS None MKL Accelerate FFTW)

set(FFT_BACKENDS None MKL Accelerate FFTW)
if(NOT FFT_BACKEND IN_LIST FFT_BACKENDS)
    message(FATAL_ERROR "Invalid FFT backend specified")
endif()

set(FWHT_BACKEND "None" CACHE STRING "Which FWHT backend to use")
set_property(CACHE FWHT_BACKEND PROPERTY STRINGS None FXT FFHT)

set(FWHT_BACKENDS None FXT FFHT)
if(NOT FWHT_BACKEND IN_LIST FWHT_BACKENDS)
    message(FATAL_ERROR "Invalid FWHT backend specified")
endif()

add_library(libcompmatmul INTERFACE hashing.hpp fft.hpp matmul_fft.hpp fwht.hpp matmul_fwht.hpp matmul.hpp)

if(FFT_BACKEND STREQUAL "None")
    message(WARNING "No FFT backend specified, disabling FFT support")
elseif(FFT_BACKEND STREQUAL "MKL")
    if(NOT BLAS_BACKEND STREQUAL "MKL")
        message(FATAL_ERROR "MKL FFT backend requires MKL BLAS backend")
    endif()
    target_compile_definitions(libcompmatmul INTERFACE COMPMATMUL_ENABLE_FFT COMPMATMUL_USE_MKL_FFT)
elseif(FFT_BACKEND STREQUAL "FFTW")
    find_package(FFTW3 CONFIG REQUIRED)
    find_package(FFTW3f CONFIG REQUIRED PATHS "${FFTW3_DIR}" NO_DEFAULT_PATH)
    target_include_directories(libcompmatmul INTERFACE ${FFTW3_INCLUDE_DIRS} ${FFTW3f_INCLUDE_DIRS})
    target_link_directories(libcompmatmul INTERFACE ${FFTW3_LIBRARY_DIRS} ${FFTW3f_LIBRARY_DIRS})
    target_link_libraries(libcompmatmul INTERFACE ${FFTW3_LIBRARIES} ${FFTW3f_LIBRARIES})
    target_compile_definitions(libcompmatmul INTERFACE COMPMATMUL_ENABLE_FFT COMPMATMUL_USE_FFTW_FFT)
elseif(FFT_BACKEND STREQUAL "Accelerate")
    if(APPLE)
        if(NOT BLAS_BACKEND STREQUAL "Accelerate")
            message(FATAL_ERROR "Accelerate FFT backend requires Accelerate BLAS backend")
        endif()
        target_compile_definitions(libcompmatmul INTERFACE COMPMATMUL_ENABLE_FFT COMPMATMUL_USE_ACCELERATE_FFT)
    else()
        message(FATAL_ERROR "Accelerate backend is only supported on macOS")
    endif()
else()
    message(FATAL_ERROR "Invalid FFT backend specified")
endif()

if(FWHT_BACKEND STREQUAL "None")
    message(WARNING "No FWHT backend specified, disabling FWHT support")
elseif(FWHT_BACKEND STREQUAL "FXT")
    target_compile_definitions(libcompmatmul INTERFACE COMPMATMUL_ENABLE_FWHT COMPMATMUL_USE_FXT_FWHT)
    find_path(FXT_INCLUDE_DIR NAMES fxttypes.h PATH_SUFFIXES fxt)
    if(NOT FXT_INCLUDE_DIR)
        message(FATAL_ERROR "Could not find FXT include directory")
    endif()
target_include_directories(libcompmatmul INTERFACE "${FXT_INCLUDE_DIR}")
elseif(FWHT_BACKEND STREQUAL "FFHT")
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
        target_compile_definitions(libcompmatmul INTERFACE COMPMATMUL_ENABLE_FWHT COMPMATMUL_USE_FFHT_FWHT)
    else()
        message(FATAL_ERROR "FFHT backend is only supported on x86_64 architectures")
    endif()
else()
    message(FATAL_ERROR "Invalid FWHT backend specified")
endif()

# we are only adding this here at this step to make sure that FFTW is found before MKL
target_link_libraries(libcompmatmul INTERFACE libarray)


if (CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(libcompmatmul INTERFACE LIBARRAY_DEBUG COMPMATMUL_DEBUG)
  target_compile_options(libcompmatmul INTERFACE -Wall -Wextra -pedantic -g -fsanitize=address)
elseif (CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_definitions(libcompmatmul INTERFACE NDEBUG)
  target_compile_options(libcompmatmul INTERFACE -Wall -Wextra -pedantic -O3)
else()
    message(FATAL_ERROR "Invalid build type")
endif()

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    target_compile_options(libarray INTERFACE -march=${ARCH} -mtune=${TUNE})
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    target_compile_options(libarray INTERFACE -march=${ARCH} -mcpu=${CPU} -mtune=${TUNE})
else()
    message(FATAL_ERROR "Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()

find_package(OpenMP REQUIRED)

target_link_libraries(libcompmatmul INTERFACE OpenMP::OpenMP_CXX)
# target_link_options(libcompmatmul INTERFACE -fopenmp)

if(BUILD_DOCS)
    find_package(Doxygen REQUIRED)
    set(DOXYGEN_GENERATE_HTML YES)
    set(DOXYGEN_GENERATE_LATEX NO)
    set(DOXYGEN_EXTRACT_ALL NO)
    set(DOXYGEN_PREDEFINED DOXYGEN)
    set(DOXYGEN_FILE_PATTERNS *.hpp)
    doxygen_add_docs(docs ${PROJECT_SOURCE_DIR})
endif()

if(BUILD_TESTS)
    find_package(Catch2 3 REQUIRED)
    add_executable(tests tests.cpp)
    target_link_libraries(tests PRIVATE libcompmatmul Catch2::Catch2WithMain)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_link_options(tests PRIVATE -fsanitize=address)
    endif()

    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # suppresses warnings originating from Catch2 headers when compiling 
        # with Clang
        target_compile_options(tests PRIVATE -Wno-c2y-extensions)
    endif()
endif()

if(BUILD_MODULE)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(compmatmul MODULE compmatmul.cpp)
    target_link_libraries(compmatmul PRIVATE libcompmatmul)
    install(TARGETS compmatmul DESTINATION ${Python_SITEARCH})
endif()

if(BUILD_BENCHMARK)
    add_executable(benchmark benchmark.cpp)
    target_include_directories(benchmark PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tclap/include)
    target_link_libraries(benchmark PRIVATE libcompmatmul)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_link_options(benchmark PRIVATE -fsanitize=address)
    endif()
    install(TARGETS benchmark DESTINATION bin)
endif()

if(BUILD_MULTIPLY)
    find_package(ZLIB)
    add_executable(multiply multiply.cpp cnpy/cnpy.cpp)
    target_include_directories(multiply PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tclap/include)
    target_link_libraries(multiply PRIVATE libcompmatmul ZLIB::ZLIB)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_link_options(multiply PRIVATE -fsanitize=address)
    endif()
    install(TARGETS multiply DESTINATION bin)
endif()

if(BUILD_SYNTHETIC)
    add_executable(synthetic synthetic.cpp)
    target_include_directories(synthetic PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tclap/include)
    target_link_libraries(synthetic PRIVATE libcompmatmul)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_link_options(synthetic PRIVATE -fsanitize=address)
    endif()
    install(TARGETS synthetic DESTINATION bin)
endif()
